567 void Breakout::collision_system() const
568 {
569 Entity ball = find_first(ObjectKind::Ball);
570 Entity state = find_first(ObjectKind::State);
571
572 if (ball.eof() || state.eof()) {
573 return;
574 }
575
576 auto& ballGameplay = ball.get<Gameplay>();
577 const auto& ballCollider = ball.get<Collider>();
578 auto& stateGameplay = state.get<Gameplay>();
579
580 if (ballGameplay.ballAttached ||
581 stateGameplay.gameOver ||
582 stateGameplay.won)
583 {
584 return;
585 }
586
587 constexpr int MAX_CONTACTS = 16;
588 b2ContactData contacts[MAX_CONTACTS];
589
590 const int capacity =
591 b2Body_GetContactCapacity(ballCollider.body);
592
593 if (capacity <= 0) {
594 return;
595 }
596
597 const int count =
598 b2Body_GetContactData(
599 ballCollider.body,
600 contacts,
601 std::min(MAX_CONTACTS, capacity)
602 );
603
604 for (int i = 0; i < count; ++i) {
605 const b2ContactData& contact = contacts[i];
606
607 if (contact.manifold.pointCount == 0) {
608 continue;
609 }
610
611 const b2BodyId bodyA =
612 b2Shape_GetBody(contact.shapeIdA);
613
614 const b2BodyId bodyB =
615 b2Shape_GetBody(contact.shapeIdB);
616
617 const b2BodyId otherBody =
618 same_body(bodyA, ballCollider.body)
619 ? bodyB
620 : bodyA;
621
622 Entity other =
623 find_entity_by_body(otherBody);
624
625 if (other.eof()) {
626 continue;
627 }
628
629 auto& otherGameplay =
630 other.get<Gameplay>();
631
632 ++ballGameplay.hitCounter;
633
634 if (otherGameplay.kind == ObjectKind::Brick) {
635 stateGameplay.score += otherGameplay.brickValue;
636 --stateGameplay.bricksLeft;
637
638 const auto& brickCollider =
639 other.get<Collider>();
640
641 if (b2Body_IsValid(brickCollider.body)) {
642 b2DestroyBody(brickCollider.body);
643 }
644
645 other.destroy();
646 break;
647 }
648
649 if (otherGameplay.kind == ObjectKind::Paddle) {
650 const float ballX =
651 ball.get<Transform>().p.x;
652
653 const float paddleX =
654 other.get<Transform>().p.x;
655
656 const float normalized =
657 (ballX - paddleX) /
658 (PADDLE_W / 2.0f);
659
660 b2Vec2 velocity =
661 b2Body_GetLinearVelocity(
662 ballCollider.body
663 );
664
665 velocity.x =
666 normalized *
667 BALL_SPEED_X /
668 BOX_SCALE;
669
670 velocity.y =
671 -std::fabs(velocity.y);
672
673 b2Body_SetLinearVelocity(
674 ballCollider.body,
675 velocity
676 );
677
678 break;
679 }
680 }
681 b2Vec2 velocity =
682 b2Body_GetLinearVelocity(ballCollider.body);
683
684 const float speed =
685 std::sqrt(
686 velocity.x * velocity.x +
687 velocity.y * velocity.y
688 );
689
690 if (speed < BALL_MIN_SPEED && speed > 0.001f) {
691 const float factor =
692 BALL_MIN_SPEED / speed;
693
694 velocity.x *= factor;
695 velocity.y *= factor;
696
697 b2Body_SetLinearVelocity(
698 ballCollider.body,
