673 void Breakout::collision_system() const
674 {
675 Entity ball = find_first<BallTag>();
676 Entity state = find_first<StateTag>();
677
678 if (ball.eof() || state.eof()) {
679 return;
680 }
681
682 auto& ballState = ball.get<BallState>();
683 const auto& ballCollider = ball.get<Collider>();
684 auto& gameState = state.get<GameState>();
685
686 if (ballState.attached ||
687 gameState.gameOver ||
688 gameState.won)
689 {
690 return;
691 }
692
693 constexpr int MAX_CONTACTS = 16;
694 b2ContactData contacts[MAX_CONTACTS];
695
696 const int capacity =
697 b2Body_GetContactCapacity(ballCollider.body);
698
699 if (capacity <= 0) {
700 return;
701 }
702
703 const int count =
704 b2Body_GetContactData(
705 ballCollider.body,
706 contacts,
707 std::min(MAX_CONTACTS, capacity)
708 );
709
710 for (int i = 0; i < count; ++i) {
711 const b2ContactData& contact = contacts[i];
712
713 if (contact.manifold.pointCount == 0) {
714 continue;
715 }
716
717 const b2BodyId bodyA =
718 b2Shape_GetBody(contact.shapeIdA);
719
720 const b2BodyId bodyB =
721 b2Shape_GetBody(contact.shapeIdB);
722
723 const b2BodyId otherBody =
724 same_body(bodyA, ballCollider.body)
725 ? bodyB
726 : bodyA;
727
728 Entity other =
729 find_entity_by_body(otherBody);
730
731 if (other.eof()) {
732 continue;
733 }
734
735 ++ballState.hitCounter;
736
737 if (other.has<BrickTag>()) {
738 gameState.score += other.get<BrickData>().value;
739 --gameState.bricksLeft;
740
741 const auto& brickCollider =
742 other.get<Collider>();
743
744 if (b2Body_IsValid(brickCollider.body)) {
745 b2DestroyBody(brickCollider.body);
746 }
747
748 other.destroy();
749 break;
750 }
751
752 if (other.has<PaddleTag>()) {
753 const float ballX =
754 ball.get<Transform>().p.x;
755
756 const float paddleX =
757 other.get<Transform>().p.x;
758
759 const float normalized =
760 (ballX - paddleX) /
761 (PADDLE_W / 2.0f);
762
763 b2Vec2 velocity =
764 b2Body_GetLinearVelocity(
765 ballCollider.body
766 );
767
768 velocity.x =
769 normalized *
770 BALL_SPEED_X /
771 BOX_SCALE;
772
773 velocity.y =
774 -std::fabs(velocity.y);
775
776 b2Body_SetLinearVelocity(
777 ballCollider.body,
778 velocity
779 );
780
781 break;
782 }
783 }
784 b2Vec2 velocity =
785 b2Body_GetLinearVelocity(ballCollider.body);
786
787 const float speed =
788 std::sqrt(
789 velocity.x * velocity.x +
790 velocity.y * velocity.y
791 );
792
793 if (speed < BALL_MIN_SPEED && speed > 0.001f) {
794 const float factor =
795 BALL_MIN_SPEED / speed;
796
797 velocity.x *= factor;
798 velocity.y *= factor;
799
800 b2Body_SetLinearVelocity(
801 ballCollider.body,
802 velocity
803 );
804 }
805 }
